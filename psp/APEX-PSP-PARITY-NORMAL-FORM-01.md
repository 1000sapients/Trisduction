## Appendix A · Parity_Normal_Form.lean

```lean
/-
# PARITY NORMAL FORM FOR BARRIERS · AND THE WIDTH LAW · Parity_Normal_Form.lean
core Lean 4 v4.19.0 · no Mathlib · no sorry · no axiom declared.
A method class reads a target d : α → Bool only through a record ρ : α → β.
(A) The barrier criterion: d is unreadable from ρ iff ρ collapses a pair d splits.
(B) The normal form: every such barrier is a parity barrier, an involution that fixes the
    record and flips the target; and every parity barrier is such a barrier.
(C) The width law: a barrier is at least one bit wide; it is exactly one bit wide iff the
    admissible family is a complement pair {d, ¬d}, and then any odd witness closes it with
    a unique calibration. The unconstrained odd family on k orbits is 2^k wide, k bits.
ΔM = 0: every step is classical, elementary, and definitional.
-/

namespace PNF

universe u v
variable {α : Type u} {β : Type v}

/-! ## A · The barrier criterion -/

def Factors (ρ : α → β) (d : α → Bool) : Prop := ∃ g : β → Bool, ∀ z, g (ρ z) = d z
def Separates (ρ : α → β) (d : α → Bool) : Prop := ∃ x y, ρ x = ρ y ∧ d x ≠ d y

theorem wall_of_sep {ρ : α → β} {d : α → Bool} : Separates ρ d → ¬ Factors ρ d := by
  intro ⟨x, y, hρ, hd⟩ ⟨g, hg⟩
  exact hd (by rw [← hg x, ← hg y, hρ])

open Classical in
theorem factors_of_not_sep {ρ : α → β} {d : α → Bool} (h : ¬ Separates ρ d) : Factors ρ d := by
  refine ⟨fun b => if ∃ x, ρ x = b ∧ d x = true then true else false, fun z => ?_⟩
  by_cases hz : d z = true
  · simp only [hz]; exact if_pos ⟨z, rfl, hz⟩
  · have hz' : d z = false := by cases e : d z <;> simp_all
    rw [hz']
    apply if_neg
    intro ⟨x, hx, hdx⟩
    exact h ⟨x, z, hx, by rw [hdx, hz']; decide⟩

theorem barrier_criterion (ρ : α → β) (d : α → Bool) : Factors ρ d ↔ ¬ Separates ρ d :=
  ⟨fun hf hs => wall_of_sep hs hf, factors_of_not_sep⟩

/-! ## B · The parity normal form -/

def swap [DecidableEq α] (x y : α) (z : α) : α := if z = x then y else if z = y then x else z

theorem swap_involution [DecidableEq α] {x y : α} (hxy : x ≠ y) :
    ∀ z, swap x y (swap x y z) = z := by
  intro z
  by_cases hzx : z = x
  · simp [swap, hzx, Ne.symm hxy]
  · by_cases hzy : z = y
    · simp [swap, hzy, Ne.symm hxy]
    · simp [swap, hzx, hzy]

theorem swap_record_even [DecidableEq α] {ρ : α → β} {x y : α} (hxy : x ≠ y) (hρ : ρ x = ρ y) :
    ∀ z, ρ (swap x y z) = ρ z := by
  intro z
  by_cases hzx : z = x
  · simp [swap, hzx, hρ]
  · by_cases hzy : z = y
    · simp [swap, hzy, Ne.symm hxy, hρ]
    · simp [swap, hzx, hzy]

/-- A parity barrier: an involution that fixes the record and flips the target somewhere. -/
def ParityBarrier (ρ : α → β) (d : α → Bool) : Prop :=
  ∃ τ : α → α, (∀ z, τ (τ z) = z) ∧ (∀ z, ρ (τ z) = ρ z) ∧ ∃ x, d (τ x) ≠ d x

/-- Every separating pair is a parity barrier. -/
theorem parity_of_sep [DecidableEq α] {ρ : α → β} {d : α → Bool} :
    Separates ρ d → ParityBarrier ρ d := by
  intro ⟨x, y, hρ, hd⟩
  have hxy : x ≠ y := fun h => hd (h ▸ rfl)
  refine ⟨swap x y, swap_involution hxy, swap_record_even hxy hρ, x, ?_⟩
  have : swap x y x = y := by simp [swap]
  rw [this]; exact fun h => hd h.symm

/-- Every parity barrier separates a pair. -/
theorem sep_of_parity {ρ : α → β} {d : α → Bool} : ParityBarrier ρ d → Separates ρ d := by
  intro ⟨τ, _, hρ, x, hx⟩
  exact ⟨τ x, x, hρ x, hx⟩

/-- THE PARITY NORMAL FORM. A target is unreadable from a record iff there is an involution
    fixing the record and flipping the target: every barrier of this kind is a parity barrier. -/
theorem parity_normal_form [DecidableEq α] (ρ : α → β) (d : α → Bool) :
    ¬ Factors ρ d ↔ ParityBarrier ρ d := by
  constructor
  · intro h
    apply parity_of_sep
    exact Classical.byContradiction (fun hs => h (factors_of_not_sep hs))
  · intro hp; exact wall_of_sep (sep_of_parity hp)

/-! ## C · The width law -/

def shift (d : α → Bool) (c : Bool) : α → Bool := fun x => xor (d x) c

theorem shift_false (d : α → Bool) : shift d false = d := by
  funext x; simp [shift]

theorem shift_true (d : α → Bool) : shift d true = fun x => !d x := by
  funext x; simp [shift]

/-- A barrier is at least one bit wide: the record-preserving symmetry carries the target to a
    second, distinct target that the record cannot tell apart from it. -/
theorem width_at_least_one [DecidableEq α] {ρ : α → β} {d : α → Bool} (h : ¬ Factors ρ d) :
    ∃ τ : α → α, (∀ z, ρ (τ z) = ρ z) ∧ (fun z => d (τ z)) ≠ d := by
  obtain ⟨τ, _, hρ, x, hx⟩ := (parity_normal_form ρ d).1 h
  exact ⟨τ, hρ, fun e => hx (congrFun e x)⟩

/-- The complement pair of d. -/
def ComplementPair (d : α → Bool) (e : α → Bool) : Prop := e = d ∨ e = fun x => !d x

/-- On a nonempty domain the complement pair is exactly the image of one bit: e is in the pair
    iff e = d xor c for some c, and c is unique. -/
theorem complement_pair_is_one_bit (d : α → Bool) (x₀ : α) :
    (∀ e, ComplementPair d e ↔ ∃ c, e = shift d c) ∧
    (∀ c₁ c₂, shift d c₁ = shift d c₂ → c₁ = c₂) := by
  refine ⟨fun e => ⟨?_, ?_⟩, ?_⟩
  · intro h; cases h with
    | inl h => exact ⟨false, by rw [h, shift_false]⟩
    | inr h => exact ⟨true, by rw [h, shift_true]⟩
  · intro ⟨c, hc⟩; cases c
    · exact Or.inl (by rw [hc, shift_false])
    · exact Or.inr (by rw [hc, shift_true])
  · intro c₁ c₂ h
    have := congrFun h x₀
    simp only [shift] at this
    cases hd : d x₀ <;> cases c₁ <;> cases c₂ <;> simp_all

/-- The two members of the pair are distinct on a nonempty domain: the width is one bit and not zero. -/
theorem complement_pair_distinct (d : α → Bool) (x₀ : α) : d ≠ fun x => !d x := by
  intro h
  have h1 : d x₀ = !d x₀ := congrFun h x₀
  generalize d x₀ = b at h1
  cases b <;> cases h1

/-- THE WIDTH LAW, converse. A nonempty admissible family closed under complement in which any
    two members are equal or complementary is exactly the complement pair of any of its members:
    those are the families one bit wide. -/
theorem one_bit_iff_complement_pair (A : (α → Bool) → Prop) (d : α → Bool) (hd : A d) :
    ((∀ e, A e → A (fun x => !e x)) ∧ (∀ e e', A e → A e' → e' = e ∨ e' = fun x => !e x)) ↔
    (∀ e, A e ↔ ComplementPair d e) := by
  constructor
  · intro ⟨hclosed, hpair⟩ e
    constructor
    · intro he; exact hpair d e hd he
    · intro h; cases h with
      | inl h => rw [h]; exact hd
      | inr h => rw [h]; exact hclosed d hd
  · intro h
    refine ⟨fun e he => ?_, fun e e' he he' => ?_⟩
    · have := (h e).1 he
      apply (h _).2
      cases this with
      | inl h1 => exact Or.inr (by rw [h1])
      | inr h1 => left; rw [h1]; funext x; simp
    · have h1 := (h e).1 he; have h2 := (h e').1 he'
      cases h1 with
      | inl h1 => cases h2 with
        | inl h2 => left; rw [h1, h2]
        | inr h2 => right; rw [h1, h2]
      | inr h1 => cases h2 with
        | inl h2 => right; rw [h1, h2]; funext x; simp
        | inr h2 => left; rw [h1, h2]

/-- The crossing on the pair: an odd witness s with s = d xor c₀ fixes the calibration uniquely,
    so one supplied bit closes a one-bit barrier. -/
theorem odd_witness_closes (d s : α → Bool) (x₀ : α) (c c' : Bool)
    (hc : s = shift d c) (hc' : s = shift d c') : c = c' :=
  (complement_pair_is_one_bit d x₀).2 c c' (hc.symm.trans hc')

/-! ## C′ · The unconstrained odd family is 2^k wide: executed on k = 1 to 6 orbits
  Points 0 .. 2k−1, the involution swaps 2i and 2i+1, a target is a bitmask; wholly odd means
  every pair is split. The count is 2^k: k bits, one bit only at k = 1. -/

def whollyOddMask (k m : Nat) : Bool :=
  (List.range k).all (fun i => m.testBit (2 * i) != m.testBit (2 * i + 1))

def oddCount (k : Nat) : Nat := ((List.range (2 ^ (2 * k))).filter (whollyOddMask k)).length

theorem odd_family_k1 : oddCount 1 = 2 := by decide
theorem odd_family_k2 : oddCount 2 = 4 := by decide
theorem odd_family_k3 : oddCount 3 = 8 := by decide
theorem odd_family_k4 : oddCount 4 = 16 := by decide

set_option maxRecDepth 100000 in
/-- The parity frame of the barrier census (K4): six pairs, 2^6 = 64 wholly odd targets,
    six bits unconstrained; the admissible row family {d₀, ¬d₀} is one bit. -/
theorem odd_family_k6 : oddCount 6 = 64 := by decide

/-! ## D · The rows papers' claim, at its grade
  "The deficit on each row is exactly one bit" is, by the width law, equivalent to "the
  admissible family on each row is a complement pair". The equivalence is a theorem; the
  per-row hypothesis is not proved here and cannot be without the row's object. -/

theorem rows_one_bit_iff_complement_pair {R : Type} (A : R → (α → Bool) → Prop)
    (d : R → α → Bool) (hd : ∀ r, A r (d r)) :
    (∀ r, (∀ e, A r e → A r (fun x => !e x)) ∧ (∀ e e', A r e → A r e' → e' = e ∨ e' = fun x => !e x)) ↔
    (∀ r e, A r e ↔ ComplementPair (d r) e) := by
  constructor
  · intro h r; exact (one_bit_iff_complement_pair (A r) (d r) (hd r)).1 (h r)
  · intro h r; exact (one_bit_iff_complement_pair (A r) (d r) (hd r)).2 (h r)

end PNF
```
